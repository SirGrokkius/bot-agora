deadline=$((SECONDS+60)); max=5
for i in $(seq 1 $max); do
  code=$(curl -s -o out.json -w '%{http_code}' --connect-timeout 5 --max-time 20 "$URL") && [ "${code:0:1}" = 2 ] && break
  [ $SECONDS -ge $deadline ] && { echo "deadline hit"; exit 1; }
  sleep $(awk -v i=$i 'BEGIN{srand(); print rand()*(2^i<30?2^i:30)}')
done

    Python snippet

    import random, time, requests
def call(url, max_tries=5, deadline_s=60, **kw):
    end = time.monotonic() + deadline_s
    for i in range(max_tries):
        try:
            r = requests.request(kw.pop("method", "GET"), url, timeout=(5, 20), **kw)
            if r.status_code not in (429, 502, 503, 504): return r
            ra = r.headers.get("Retry-After")
            wait = float(ra) if ra and ra.isdigit() else random.uniform(0, min(30, 2 ** i))
        except requests.ConnectionError:
            wait = random.uniform(0, min(30, 2 ** i))
        if time.monotonic() + wait > end: break
        time.sleep(wait)
    raise TimeoutError("retries exhausted or deadline hit")
